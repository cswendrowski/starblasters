# Parallax V4 Implementation Plan

> **For agentic workers:** Execute this plan task-by-task using code-editor subagents. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the flat Node2D `galaxy_backdrop.gd` with a 7-scene per-layer CanvasLayer architecture where each layer has its own CanvasModulate for independent color grading.

**Architecture:** Coordinator scene instantiates 7 CanvasLayer scenes. Each layer exports `scroll_rate: float` and `modulate_color: Color`. Coordinator drives `layer.offset.y` scroll and sets modulate colors from planet type. Object-level respawn (V1's reset-at-y>310 logic) stays inside each layer's script — `offset.y` alone does not wrap content.

**Tech Stack:** Godot 4.6.3 GDScript, `CanvasLayer`, `CanvasModulate`, `starfield.gdshader`, `nebula2.gdshader`, PixelPlanets scenes, `GPUParticles2D`. New scenes under `scenes/parallax/layers/`, new scripts under `scripts/parallax/`.

**Key invariants:**
- `_apply_pixel_parity(p, size)` MUST be called AFTER `add_child(p)` on planet nodes (fix from commit 7d834da — do not regress this)
- All callers switch from `set_script(BACKDROP_SCRIPT)` to `preload("res://scenes/parallax/backdrop_coordinator.tscn").instantiate()`
- `galaxy_backdrop.gd` stays on disk as V1 archive until V4 is confirmed working
- Build tuner LAST — system must be visually confirmed first

The stars layer contains two `ColorRect`s each with `starfield.gdshader`, inside two `Parallax2D` nodes at different `scroll_scale.y` values for depth separation. The coordinator nudges the Parallax2D `scroll_offset` each frame.

- [ ] **Step 1: Write `layer_stars.gd`**

```gdscript
extends ParallaxLayerBase

@export var far_density: float = 32.0
@export var far_threshold: float = 0.96
@export var near_density: float = 20.0
@export var near_threshold: float = 0.94
@export var near_brightness: float = 1.15

# Far starfield scrolls at 0.02×, near at 0.08× of drift_speed.
const FAR_RATE  := 0.02
const NEAR_RATE := 0.08

var _far_para: Node = null
var _near_para: Node = null
var _scroll_accum_far: float  = 0.0
var _scroll_accum_near: float = 0.0


func _ready() -> void:
    super._ready()
    _far_para  = $FarStars
    _near_para = $NearStars


func _on_scrolled() -> void:
    pass  # stars driven by _process, not by offset.y directly


func _process(delta: float) -> void:
    # Stars use shader scroll via Parallax2D scroll_offset, not CanvasLayer offset.y.
    # offset.y on this layer stays 0.
    pass


# Called by coordinator with drift_speed * delta each frame.
func scroll_stars(drift_delta: float) -> void:
    _scroll_accum_far  += drift_delta * FAR_RATE
    _scroll_accum_near += drift_delta * NEAR_RATE
    if _far_para != null:
        _far_para.scroll_offset = Vector2(0, _scroll_accum_far)
    if _near_para != null:
        _near_para.scroll_offset = Vector2(0, _scroll_accum_near)


func _on_reset() -> void:
    _scroll_accum_far  = 0.0
    _scroll_accum_near = 0.0
    if _far_para != null:
        _far_para.scroll_offset = Vector2.ZERO
    if _near_para != null:
        _near_para.scroll_offset = Vector2.ZERO
```

    mat.set_shader_parameter("brightness", brightness)
    mat.set_shader_parameter("scroll_speed", 0.0)
    rect.material = mat

- [ ] **Step 1: Write `layer_stellar.gd`**

```gdscript
extends ParallaxLayerBase

const RESET_THRESHOLD := 340.0   # y > 340 → requeue at top

@export var asteroid_count: int = 4
@export var asteroid_min_size: float = 8.0
@export var asteroid_max_size: float = 20.0
@export var nebula_enabled: bool = false
@export var nebula_alpha: float = 0.18
@export var mine_count: int = 0   # >0 for bg mine hazard decoration

const ASTEROID_SCENE = preload("res://Planets/Asteroids/Asteroid.tscn")
const NEBULA2_SHADER  = preload("res://graphics/shaders/nebula2.gdshader")

var _objects: Array = []   # array of {node, base_x, base_y, size}
var _nebula_rect: ColorRect = null
var _nebula_scroll: float = 0.0
const NEBULA_TILE: float = 270.0


func _ready() -> void:
    super._ready()


func populate(rng: RandomNumberGenerator, drift_speed_ref: float) -> void:
    _clear()
    for _i in asteroid_count:
        _spawn_asteroid(rng)
    if nebula_enabled:
        _spawn_nebula(rng)
    if mine_count > 0:
        _spawn_bg_mines(rng, mine_count)


func _clear() -> void:
    for entry in _objects:
        if is_instance_valid(entry.node):
            entry.node.queue_free()
    _objects.clear()
    if _nebula_rect and is_instance_valid(_nebula_rect):
        _nebula_rect.queue_free()
    _nebula_rect = null


func _spawn_asteroid(rng: RandomNumberGenerator) -> void:
    var a = ASTEROID_SCENE.instantiate()
    var sz := rng.randf_range(asteroid_min_size, asteroid_max_size)
    var sf := sz / 100.0
    a.scale = Vector2(sf, sf)
    a.position = Vector2(rng.randf_range(16, 464), rng.randf_range(-270, 0))
    add_child(a)
    _apply_asteroid_pixel_parity(a, sz)
    _objects.append({"node": a, "size": sz})


func _apply_asteroid_pixel_parity(a: Node, sz: float) -> void:
    # Minimal parity: set pixel count to match display size at pixel_density=1.
    var px := maxf(sz, 16.0)
    if a.has_method("set_pixels"):
        a.set_pixels(px)


func _spawn_nebula(rng: RandomNumberGenerator) -> void:
    _nebula_rect = ColorRect.new()
    _nebula_rect.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
    var mat := ShaderMaterial.new()
    mat.shader = NEBULA2_SHADER
    mat.set_shader_parameter("alpha",     nebula_alpha)
    mat.set_shader_parameter("seed_val",  float(rng.randi() % 10000))
    _nebula_rect.material = mat
    add_child(_nebula_rect)


func _spawn_bg_mines(rng: RandomNumberGenerator, count: int) -> void:
    # Decorative only — no collision.
    for _i in count:
        var s := Sprite2D.new()
        # Use a simple placeholder for now; swap in mine texture when available.
        s.position = Vector2(rng.randf_range(80, 400), rng.randf_range(-270, 0))
        s.scale = Vector2(0.4, 0.4)
        s.modulate = Color(0.5, 0.5, 0.5, 0.7)
        add_child(s)
        _objects.append({"node": s, "size": 8.0})


func _on_scrolled() -> void:
    # Wrap objects that drift past the bottom back to the top.
    for entry in _objects:
        var n: Node = entry.node
        if not is_instance_valid(n):
            continue
        var world_y: float = offset.y + n.position.y
        if world_y > RESET_THRESHOLD:
            n.position.y = rng_respawn_y()
            n.position.x = randf_range(16, 464)


func rng_respawn_y() -> float:
    # Respawn just above the viewport top, accounting for current offset.
    return -offset.y - randf_range(0, 270)


func randf_range(a: float, b: float) -> float:
    return a + randf() * (b - a)


func _on_reset() -> void:
    _clear()


func _process(_delta: float) -> void:
    if _nebula_rect and is_instance_valid(_nebula_rect):
        _nebula_scroll += offset.y
        if _nebula_rect.material is ShaderMaterial:
            (_nebula_rect.material as ShaderMaterial).set_shader_parameter(
                "scroll_offset", Vector2(0, offset.y / NEBULA_TILE)
            )
```

### Task 6: Coordinator

**Files:**

The coordinator is the direct replacement for `galaxy_backdrop.gd`. It must expose the same `@export` surface and populate itself from `Run.current_stellar`.

- [ ] **Step 1: Write `backdrop_coordinator.gd`**

```gdscript
extends Node2D

# Drop-in replacement for galaxy_backdrop.gd.
# Instantiate backdrop_coordinator.tscn instead of set_script(BACKDROP_SCRIPT).

@export var drift_speed: float = 22.0
@export var planet_size: float = 240.0
@export var planet_size_variance: float = 0.35
@export var tint_alpha: float = 0.09
@export var use_warp_streaks: bool = true
@export var warp_streak_count: int = 14
@export var warp_streak_speed: float = 750.0
@export var forced_planet_idx: int = -1
@export var pixel_density: float = 1.0

const PLANET_TINT := {
    0: Color(1.0, 0.45, 0.20),  # LavaWorld — orange
    1: Color(0.60, 0.85, 1.0),  # IceWorld — icy blue
    2: Color(0.85, 0.65, 0.40), # DryTerran — tan
    3: Color(0.65, 0.80, 0.55), # GasPlanet — muted green
    4: Color(0.70, 0.70, 0.75), # NoAtmosphere — grey
    5: Color(0.45, 0.75, 0.55), # LandMasses — earthy green
    6: Color(0.10, 0.10, 0.15), # BlackHole — near-black (skip tint)
    7: Color(0.55, 0.45, 0.80), # Galaxy — purple
    8: Color(1.00, 0.95, 0.75), # Star — warm white (skip tint)
}

var _layer_stars: ParallaxLayerBase = null
var _layer_planet: ParallaxLayerBase = null
var _layer_stellar_far: ParallaxLayerBase = null
var _layer_stellar_mid: ParallaxLayerBase = null
var _layer_stellar_near: ParallaxLayerBase = null
var _layer_streaks: ParallaxLayerBase = null
var _layer_composite: CanvasLayer = null

var _scroll_layers: Array = []  # layers that get offset.y scroll each frame


func _ready() -> void:
    _layer_stars         = $LayerStars
    _layer_planet        = $LayerPlanet
    _layer_stellar_far   = $LayerStellarFar
    _layer_stellar_mid   = $LayerStellarMid
    _layer_stellar_near  = $LayerStellarNear
    _layer_streaks       = $LayerStreaks
    _layer_composite     = $LayerComposite

    # Stars use their own scroll (Parallax2D) — don't add to offset.y list.
    _scroll_layers = [_layer_stellar_far, _layer_stellar_mid, _layer_stellar_near]

    _populate()


func _populate() -> void:
    var run_node := get_node_or_null("/root/Run")
    var stellar: Dictionary = {}
    if run_node and "current_stellar" in run_node:
        stellar = run_node.current_stellar

    var rng := RandomNumberGenerator.new()
    var seed_val: int = 12345
    if run_node:
        seed_val = int(run_node.run_seed) + int(run_node.get("sectors_cleared", 0)) * 9973
    rng.seed = abs(seed_val)

    # Planet
    var planet_idx: int = forced_planet_idx
    if planet_idx < 0:
        planet_idx = stellar.get("planet_idx", rng.randi() % 9)
    var size_mult := rng.randf_range(1.0 - planet_size_variance, 1.0 + planet_size_variance)
    var actual_size := planet_size * size_mult
    if _layer_planet and _layer_planet.has_method("spawn_planet"):
        _layer_planet.call("spawn_planet", planet_idx, actual_size, rng, "")

    # Tints
    _apply_tints(planet_idx)

    # Stellar populate
    for layer in [_layer_stellar_far, _layer_stellar_mid, _layer_stellar_near]:
        if layer and layer.has_method("populate"):
            layer.call("populate", rng, drift_speed)

    # Streaks
    if _layer_streaks and "streak_count" in _layer_streaks:
        _layer_streaks.set("streak_count", warp_streak_count)
        _layer_streaks.set("streak_speed", warp_streak_speed)
        _layer_streaks.set("enabled", use_warp_streaks)

    # Composite grade + vignette
    _setup_composite(planet_idx)


func _apply_tints(planet_idx: int) -> void:
    var base_tint: Color = PLANET_TINT.get(planet_idx, Color.WHITE)
    var skip_tint: bool = planet_idx == 6 or planet_idx == 8  # BlackHole, Star
    var grade_color: Color = Color.WHITE.lerp(base_tint, tint_alpha) if not skip_tint else Color.WHITE

    _set_layer_modulate(_layer_stellar_far,  base_tint.lerp(Color.WHITE, 0.75))
    _set_layer_modulate(_layer_stellar_mid,  base_tint.lerp(Color.WHITE, 0.60))
    _set_layer_modulate(_layer_stellar_near, base_tint.lerp(Color.WHITE, 0.45))
    # Composite CanvasModulate drives global mood
    if _layer_composite and _layer_composite.has_node("CanvasModulate"):
        (_layer_composite.get_node("CanvasModulate") as CanvasModulate).color = grade_color


func _set_layer_modulate(layer: ParallaxLayerBase, color: Color) -> void:
    if layer == null:
        return
    layer.modulate_color = color


func _setup_composite(planet_idx: int) -> void:
    if _layer_composite == null:
        return
    var grade := _layer_composite.get_node_or_null("GradeRect") as ColorRect
    if grade:
        var mat := CanvasItemMaterial.new()
        mat.blend_mode = CanvasItemMaterial.BLEND_MODE_MUL
        grade.material = mat
    var vignette := _layer_composite.get_node_or_null("Vignette") as Sprite2D
    if vignette:
        var mat2 := CanvasItemMaterial.new()
        mat2.blend_mode = CanvasItemMaterial.BLEND_MODE_MUL
        vignette.material = mat2
        # Build vignette radial gradient texture (same as V1 _spawn_vignette)
        var grad := Gradient.new()
        grad.colors = PackedColorArray([Color.WHITE, Color(0.74, 0.74, 0.74, 1)])
        var tex := GradientTexture2D.new()
        tex.gradient = grad
        tex.width = 64; tex.height = 64
        tex.fill = GradientTexture2D.FILL_RADIAL
        tex.fill_from = Vector2(0.5, 0.5)
        tex.fill_to  = Vector2(1.0, 0.5)
        vignette.texture = tex


func _process(delta: float) -> void:
    # Scroll offset-based layers
    var drift_delta := drift_speed * delta
    for layer in _scroll_layers:
        if layer != null:
            layer.scroll(drift_delta * layer.scroll_rate)
    # Stars use separate Parallax2D scroll
    if _layer_stars and _layer_stars.has_method("scroll_stars"):
        _layer_stars.call("scroll_stars", drift_delta)
```
